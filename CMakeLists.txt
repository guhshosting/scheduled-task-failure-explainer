cmake_minimum_required(VERSION 3.24)
project(scheduled-task-failure-explainer VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and the MSVC ABI toolchain are required")
endif()
option(STFE_ANALYZE "Enable MSVC code analysis" OFF)
option(STFE_ASAN "Enable address sanitizer" OFF)
option(STFE_INTEGRATION "Enable read-only existing-task integration test" OFF)
set(STFE_TEST_TASK "\\Microsoft\\Windows\\Defrag\\ScheduledDefrag" CACHE STRING "Existing task for integration")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(STFE_ASAN)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
else()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(stfe_core src/model.cpp src/acquire.cpp)
target_include_directories(stfe_core PUBLIC src)
target_compile_definitions(stfe_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(stfe_core PUBLIC /W4 /WX /utf-8 /permissive- /EHsc)
target_link_libraries(stfe_core PUBLIC taskschd wevtapi ole32 oleaut32)
if(STFE_ANALYZE)
  target_compile_options(stfe_core PUBLIC /analyze)
endif()
if(STFE_ASAN)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    message(FATAL_ERROR "Use Release or RelWithDebInfo for ASan; Debug /RTC1 is incompatible")
  endif()
  target_compile_options(stfe_core PUBLIC /fsanitize=address /Zi)
  target_link_options(stfe_core PUBLIC /INCREMENTAL:NO)
  get_filename_component(STFE_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_file(STFE_ASAN_RUNTIME clang_rt.asan_dynamic-x86_64.dll PATHS "${STFE_COMPILER_BIN}" NO_DEFAULT_PATH REQUIRED)
  add_custom_target(asan_runtime ALL COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${STFE_ASAN_RUNTIME}" "${CMAKE_BINARY_DIR}/clang_rt.asan_dynamic-x86_64.dll")
endif()
add_executable(scheduled-task-failure-explainer src/main.cpp src/version.rc src/app.manifest)
target_link_libraries(scheduled-task-failure-explainer PRIVATE stfe_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(stfe_tests tests/tests.cpp)
  target_link_libraries(stfe_tests PRIVATE stfe_core)
  add_test(NAME synthetic COMMAND stfe_tests)
  add_test(NAME cli_help COMMAND scheduled-task-failure-explainer --help)
  add_test(NAME cli_version COMMAND scheduled-task-failure-explainer --version)
  add_test(NAME cli_invalid COMMAND scheduled-task-failure-explainer --unknown)
  set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
  set_tests_properties(synthetic cli_help cli_version cli_invalid PROPERTIES TIMEOUT 15)
  if(STFE_INTEGRATION)
    add_test(NAME existing_task_read_only COMMAND scheduled-task-failure-explainer --json "${STFE_TEST_TASK}")
    set_tests_properties(existing_task_read_only PROPERTIES TIMEOUT 15)
  endif()
endif()
install(TARGETS scheduled-task-failure-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE VALIDATION.md DESTINATION share/scheduled-task-failure-explainer)
install(FILES docs/schema.json DESTINATION share/scheduled-task-failure-explainer/docs)
